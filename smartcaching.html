Personal Media Gateway Architecture & Backend Code

Reference FastAPI endpoints and yt-dlp asynchronous background workers

FastAPI Async Engine
main_gateway.py 
from fastapi import FastAPI, BackgroundTasks, HTTPException
from fastapi.responses import FileResponse
import yt_dlp
import asyncio

app = FastAPI(title="Local Media Automation Gateway")

MEDIA_DIR = "./media_library"

# Background yt-dlp task execution
def download_media_task(url: str, format_type: str):
    ydl_opts = {
        'outtmpl': f'{MEDIA_DIR}/%(title)s.%(ext)s',
        'format': 'bestvideo[ext=mp4]+bestaudio[ext=m4a]/best' if format_type == 'video' else 'bestaudio',
        'postprocessors': [{
            'key': 'FFmpegVideoConvertor',
            'preferedformat': 'mp4',
        }]
    }
    with yt_dlp.YoutubeDL(ydl_opts) as ydl:
        ydl.download([url])

# Queue video or RSS feed URL
@app.post("/api/queue")
async def add_to_queue(url: str, format_type: str, background_tasks: BackgroundTasks):
    background_tasks.add_task(download_media_task, url, format_type)
    return {"status": "queued", "url": url}

# Stream local cached media to LAN Clients
@app.get("/api/stream/{file_name}")
async def stream_media(file_name: str):
    file_path = f"{MEDIA_DIR}/{file_name}"
    return FileResponse(file_path, media_type="video/mp4")
